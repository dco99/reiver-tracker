<script>
const sports = [
  {name:"Football", wins:0, losses:0, ties:0},
  {name:"Men's Basketball", wins:0, losses:0, ties:0},
  {name:"Women's Basketball", wins:0, losses:0, ties:0},
  {name:"Baseball", wins:0, losses:0, ties:0},
  {name:"Softball", wins:0, losses:0, ties:0},
  {name:"Volleyball", wins:0, losses:0, ties:0},
  {name:"Men's Soccer", wins:0, losses:0, ties:0},
  {name:"Women's Soccer", wins:0, losses:0, ties:0},
  {name:"Track & Field", wins:0, losses:0, ties:0},
  {name:"Wrestling", wins:0, losses:0, ties:0}
];

function winPct(w,l,t){const total=w+l+t; return total===0?"0.000":(w/total).toFixed(3);}
function render(){
  const container=document.getElementById("sportsContainer");
  container.innerHTML="";
  sports.forEach((s,i)=>{
    const card=document.createElement("div"); card.className="card";
    card.innerHTML=`
      <h2>${s.name}</h2>
      <div class="record">${s.wins}-${s.losses}-${s.ties}</div>
      <div>Win %: ${winPct(s.wins,s.losses,s.ties)}</div>
      <button onclick="addWin(${i})">+ Win</button>
      <button onclick="addLoss(${i})">+ Loss</button>
      <button onclick="addTie(${i})">+ Tie</button>
    `;
    container.appendChild(card);
  });
}
function addWin(i){sports[i].wins++; render();}
function addLoss(i){sports[i].losses++; render();}
function addTie(i){sports[i].ties++; render();}
render();
</script>
